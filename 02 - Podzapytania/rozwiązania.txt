/*
1. Wyświetl nazwiska i stanowiska pracowników pracujących w tym samym dziale co
pracownik o nazwisku Miller (5 wierszy)

*/

select
	e.first_name ,
	e.last_name ,
	(select job_title from jobs where job_id=e.job_id ) jt
	from employees e
where
	e.department_id = (select department_id from employees
						where last_name = 'Miller')


/*
2. Wyświetl następujące dane o najdłużej zatrudnionym pracowniku na stanowisku
FI_ACCOUNT (1 wiersz)


*/

select
	e.first_name ,
	e.last_name ,
	e.job_id 
	from employees e
where
	e.hire_date = (select min(hire_Date) from employees
					where job_id = 'FI_ACCOUNT') 
	and  job_id = 'FI_ACCOUNT'


/*
3. Zobacz najdłużej pracujących pracowników każdego działu. (12 wierszy)


*/
explain
select
	e.last_name ,
	e.hire_date ,
	e.department_id ,
	d.department_name 
	from employees e join departments d on e.department_id = d.department_id
where
	(e.department_id , hire_Date) in (select department_id, min(hire_date)
										from employees
										group by department_id)
explain
select
	e.last_name ,
	e.hire_date ,
	e.department_id ,
	d.department_name 
	from employees e join departments d on e.department_id = d.department_id
where
	hire_Date in (select min(hire_date)
						from employees
						where department_id = d.department_id )
						
/*
4. Wyświetlaj te nazwy stanowisk, w których nie zatrudniono nikogo z działu
MARKETING. Wykorzystaj podzapytanie w podzapytaniu (17 wierszy)


*/
explain
select
	*
	from jobs
where
	job_id in (select job_id from employees
				where department_id not in (select department_id from departments
												where department_name = 'Marketing'))
						

/*
5. Wyświetl informacje o pracownikach zarabiających więcej niż średnia pensja dla ich
etatu(job_id) (43 wiersze)

*/


select 
	e.first_name ,
	e.last_name ,
	e.salary ,
	j.job_title ,
	(select avg(salary) from employees where job_id = j.job_id) srednia
	from employees e join jobs j on e.job_id = j.job_id 
where
	e.salary > (select avg(salary) from employees
					where job_id = e.job_id )
