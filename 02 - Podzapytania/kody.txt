WITH RECURSIVE hierarchia AS (
-- Kotwica: pracownicy bez managera (top-level)
		SELECT
		employee_id,
		last_name,
		manager_id,
		1 AS poziom,
		last_name::TEXT AS sciezka
		FROM employees
		WHERE manager_id IS NULL
	UNION ALL
		-- Rekurencja: dołączamy podwładnych
		SELECT
		e.employee_id,
		e.last_name,
		e.manager_id,
		h.poziom + 1,
		h.sciezka || ' > ' || e.last_name
		FROM employees e
		JOIN hierarchia h ON e.manager_id = h.employee_id)
SELECT
*
FROM hierarchia
ORDER BY sciezka;

-- Wymuszenie materializacji (zapisanie wyniku do tymczasowej struktury)
explain
WITH dane AS MATERIALIZED (
	SELECT 
	e.last_name, 
	d.department_id, 
	d.department_name 
	FROM employees e join departments d on e.department_id = d.department_id 
)
SELECT * FROM dane WHERE department_id < 100;

				
				
CREATE OR REPLACE FUNCTION get_employees_departments()
RETURNS TABLE (
    last_name      VARCHAR,
    department_id  INT,
    department_name VARCHAR
)
LANGUAGE sql
AS $$
    SELECT 
        e.last_name, 
        d.department_id, 
        d.department_name 
    FROM employees e
    JOIN departments d ON e.department_id = d.department_id;
$$;	
				

explain
SELECT * FROM get_employees_departments();


explain
WITH dane AS MATERIALIZED (
	SELECT * FROM get_employees_departments()
)
SELECT * FROM dane WHERE department_id < 100;
