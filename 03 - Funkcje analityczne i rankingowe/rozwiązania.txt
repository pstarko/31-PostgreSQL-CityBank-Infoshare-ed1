/*
1. Pokaż sumę plac dla każdego departamentu z podsumowaniem dla sum większych od 8000
(10 wierszy)

2. Zamień NULL na wartość RESULT. Użyj funkcji GROUPING (10 wierszy)

*/

select
	department_id,
	sum(salary)
	from employees e 
group by 
	rollup(department_id) 
having
	sum(salary) > 8000
	

select
	case 
		when grouping(department_id) = 1 then 'Result'
		else department_id::text
	end dept_id,
	sum(salary)
	from employees e 
group by 
	rollup(department_id) 
having
	sum(salary) > 8000
	

/*
3. Dla każdego departamentu o nr pomiędzy 10 i 50 oraz etatu pokaż sumę płacy podstawowej
wraz z podsumowaniami dla każdej podgrupy. Usuń podsumowanie dla całości(23 wiersze)

*/

select
	department_id,
	job_id,
	grouping(department_id),
	grouping(job_id),
	sum(salary)
	from employees e 
where
	department_id between 10 and 50
group by 
	cube(department_id,job_id) 
having 
	grouping(department_id) + 	grouping(job_id) <> 2
	

select
	department_id,
	job_id,
	sum(salary)
	from employees e 
where
	department_id between 10 and 50
group by 
	grouping sets((department_id,job_id), (department_id), (job_id)) 


/*
1. Zbuduj zapytanie, które dla każdego pracownika pokaże jego miejsce zatrudnienia. Następnie
utwórz ranking pracowników ze względu na zarobki bez prowizji(106 wierszy)

2. Zmodyfikuj zapytanie z p. 1 w taki sposób, aby w zbiorze wynikowym pojawiła się dodatkowa
kolumna pokazująca ranking gęsty. Usuń z rankingu gęstego pozycję 4,5,8 (103 wiersze)


3. Zmodyfikuj zapytanie z poprzedniego punktu w taki sposób, aby otrzymać dane jedynie
trzech pierwszych pracowników w rankingu pensji (weź pod uwagę ranking zwykły) ze
względu na miasto. Wykorzystaj do filtrowania danych WHERE, a nie LIMIT(17 wierszy)

*/
select 
	e.last_name,
	e.salary,
	l.city,
	rank() over (order by salary desc)
	from employees e join departments d on e.department_id=d.department_id
				     join locations l on d.location_id=l.location_id
				     
--2
select 
	t1.*
	from (
			select 
				e.last_name,
				e.salary,
				l.city,
				rank() over (order by salary desc),
				dense_rank() over (order by salary desc) drank
				from employees e join departments d on e.department_id=d.department_id
							     join locations l on d.location_id=l.location_id ) t1
where 
	t1.drank not in (4,5,8)
				     
--3
select 
	t1.*
	from (
			select 
				e.last_name,
				e.salary,
				l.city,
				rank() over (partition by l.city order by salary desc) rnk
				from employees e join departments d on e.department_id=d.department_id
							     join locations l on d.location_id=l.location_id ) t1
where 
	t1.rnk <=3			     
				     
				     
	/*
4. Wygeneruj poniższy raport z wykorzystaniem PERCENT RANK (11 wierszy).


*/
--4
select 
	t1.department_name,
	t1.ilosc_pracownikow,
	case 
		when t1.ilosc_pracownikow = (select max(t2.ilosc) from (select count(*) ilosc from employees group by department_id ) t2)
			then 'Department z największą ilością pracowników'
		when t1.ilosc_pracownikow = (select min(t2.ilosc) from (select count(*) ilosc from employees group by department_id ) t2)
			then 'Department z najmniejszą ilością pracowników'
		else t1.opis
	end result	
	from (
		select 
		    d.department_name,
			COUNT(*) ilosc_pracownikow,
		    percent_rank() over (order by count(*) DESC)*100 || ' % departamentów z większą ilością pracowników' opis
		    from employees e join departments d on e.department_id=d.department_id
		group by d.department_name ) t1
					     
/*
5. Pokaż najlepiej zarabiającego pracownika w każdym mieście (7 wierszy)



*/
select
	t1.*
	from (
		select 
			e.last_name,
			l.city,
			rank() over (partition by l.city order by salary desc) rnk
			from employees e join departments d on e.department_id=d.department_id
							 join locations l on d.location_id = l.location_id ) t1
where
	t1.rnk = 1
				
/*
7. Pokaż najlepiej i najgorzej zarabiającego pracownika. Wykorzystaj funkcję rankingową i
operator UNION ALL (2 wiersze)

*/
select
	t1.last_name ,
	t1.salary 
	from (
	select 
		last_name,
		salary,
		rank() over (order by salary desc) rk
		from employees e
	union all	
	select 
			last_name,
			salary,
			rank() over (order by salary ) rk
			from employees e ) t1
where
	t1.rk = 1
