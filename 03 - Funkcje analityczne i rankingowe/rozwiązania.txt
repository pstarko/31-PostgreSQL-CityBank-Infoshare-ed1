/*
1. Pokaż sumę plac dla każdego departamentu z podsumowaniem dla sum większych od 8000
(10 wierszy)

2. Zamień NULL na wartość RESULT. Użyj funkcji GROUPING (10 wierszy)

*/

select
	department_id,
	sum(salary)
	from employees e 
group by 
	rollup(department_id) 
having
	sum(salary) > 8000
	

select
	case 
		when grouping(department_id) = 1 then 'Result'
		else department_id::text
	end dept_id,
	sum(salary)
	from employees e 
group by 
	rollup(department_id) 
having
	sum(salary) > 8000
	

/*
3. Dla każdego departamentu o nr pomiędzy 10 i 50 oraz etatu pokaż sumę płacy podstawowej
wraz z podsumowaniami dla każdej podgrupy. Usuń podsumowanie dla całości(23 wiersze)

*/

select
	department_id,
	job_id,
	grouping(department_id),
	grouping(job_id),
	sum(salary)
	from employees e 
where
	department_id between 10 and 50
group by 
	cube(department_id,job_id) 
having 
	grouping(department_id) + 	grouping(job_id) <> 2

	job_id,
	sum(salary)
	from employees e 
where
	department_id between 10 and 50
group by 
	grouping sets((department_id,job_id), (department_id), (job_id))
